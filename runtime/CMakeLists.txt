cmake_minimum_required(VERSION 3.20)
project(QingMMD LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(SABA ${CMAKE_CURRENT_SOURCE_DIR}/deps/saba-master)
set(BULLET ${CMAKE_CURRENT_SOURCE_DIR}/deps/bullet3-2.83.7)
file(GLOB_RECURSE BULLET_SOURCE CONFIGURE_DEPENDS
  ${BULLET}/src/LinearMath/*.cpp
  ${BULLET}/src/BulletCollision/*.cpp
  ${BULLET}/src/BulletDynamics/*.cpp)
add_library(CerisBullet STATIC ${BULLET_SOURCE})
target_include_directories(CerisBullet PUBLIC ${BULLET}/src)
file(GLOB SABA_SOURCE CONFIGURE_DEPENDS ${SABA}/src/Saba/Base/*.cpp ${SABA}/src/Saba/Model/MMD/*.cpp)
add_library(CerisSaba STATIC ${SABA_SOURCE})
target_include_directories(CerisSaba PUBLIC ${SABA}/src ${SABA}/external/glm/include ${SABA}/external/spdlog/include ${SABA}/external/json/include)
target_link_libraries(CerisSaba PUBLIC CerisBullet)
target_compile_definitions(CerisSaba PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX UNICODE _UNICODE)
add_library(CerisMMDCore SHARED src/core.cpp)
target_link_libraries(CerisMMDCore PRIVATE CerisSaba)
target_include_directories(CerisMMDCore PUBLIC include)
set(UE4SS ${CMAKE_CURRENT_SOURCE_DIR}/deps/RE-UE4SS-1ceefac8b8c286f0fb3eb62b4aae075ff81df0bd)
add_library(CerisMMDBridge SHARED src/bridge.cpp src/core.cpp)
set_target_properties(CerisMMDBridge PROPERTIES OUTPUT_NAME main)
target_include_directories(CerisMMDBridge PRIVATE include generated/sdk ${UE4SS}/UE4SS/include ${UE4SS}/deps/first/String/include ${UE4SS}/deps/first/File/include ${UE4SS}/deps/first/Input/include)
target_link_libraries(CerisMMDBridge PRIVATE CerisSaba ${CMAKE_CURRENT_SOURCE_DIR}/generated/sdk/UE4SS.lib)
if(MSVC)
  target_compile_options(CerisBullet PRIVATE /W0)
  target_compile_options(CerisSaba PRIVATE /W0 /utf-8)
  target_compile_options(CerisMMDCore PRIVATE /W4 /utf-8)
  target_compile_options(CerisMMDBridge PRIVATE /W4 /utf-8)
endif()
